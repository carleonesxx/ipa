function SleepPage({ notify }: { notify: (s: string) => void }) { const [items, setItems] = useState<Array<{ id: string; durationMinutes: number; source: string; date: string }>>([]); const load = () => apiFetch('/sleep').then(data => setItems(data.items || [])).catch(() => notify('Не удалось загрузить сон')); useEffect(() => { void load() }, []); const add = async () => { const hours = window.prompt('Сколько часов сна?', '8'); if (hours === null) return; try { await apiFetch('/sleep', { method: 'POST', body: JSON.stringify({ durationMinutes: Math.round(Number(hours) * 60), source: 'manual' }) }); await load(); notify('Сон сохранён') } catch (e) { notify(e instanceof Error ? e.message : 'Не удалось сохранить сон') } }; const average = items.length ? items.reduce((sum, x) => sum + x.durationMinutes, 0) / items.length : 0; return <><PageTitle eyebrow="Сон" title="Восстановление в вашем ритме" sub="Ручные записи и данные подключённых источников." action={<button className="button primary" onClick={add}><Plus size={17} /> Добавить сон</button>} /><section className="card weekly-card"><div className="metric-grid"><MetricCard icon={<Moon size={19} />} label="Средняя длительность" value={`${(average / 60).toFixed(1)} ч`} target="По сохранённым записям" percent={0} tone="purple" /></div>{items.length ? <div className="meal-list">{items.map(item => <div className="meal-row" key={item.id}><div className="meal-info"><b>{(item.durationMinutes / 60).toFixed(1)} ч сна</b><small>{new Date(item.date).toLocaleDateString('ru-RU')} · {item.source}</small></div><button className="row-more" onClick={async () => { await apiFetch(`/sleep/${item.id}`, { method: 'DELETE' }); await load(); notify('Запись удалена') }}><X size={16} /></button></div>)}</div> : <div className="empty-state"><Moon size={22} /><b>Записей сна пока нет</b><span>Добавьте первую запись.</span></div>}</section></> }
